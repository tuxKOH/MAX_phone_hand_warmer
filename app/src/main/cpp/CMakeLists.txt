cmake_minimum_required(VERSION 3.22.1)
project(maxheater LANGUAGES CXX)
add_library(maxheater SHARED engine.cpp qnn_runner.cpp)
target_link_libraries(maxheater EGL GLESv3 android log dl)
target_compile_options(maxheater PRIVATE -Wall -Wextra -Werror=return-type)
if(QNN_SDK_ROOT AND EXISTS "${QNN_SDK_ROOT}/include/QNN/QnnInterface.h" AND ANDROID_ABI STREQUAL "arm64-v8a")
  target_include_directories(maxheater PRIVATE "${QNN_SDK_ROOT}/include/QNN")
  target_compile_definitions(maxheater PRIVATE MAX_HAS_QNN=1)
endif()
